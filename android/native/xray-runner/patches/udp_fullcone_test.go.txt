func TestAndroidUDPPreservesLargeDatagramsWithoutCopy(t *testing.T) {
    c := &udpConn{egress:make(chan []byte,1)}
    payload := bytes.Repeat([]byte{42},9000)
    c.egress <- payload
    reader := buf.NewReader(c)
    data,err := reader.ReadMultiBuffer()
    if err != nil || len(data) != 1 || !bytes.Equal(data[0].Bytes(),payload) { t.Fatal("datagram truncated") }
    if &data[0].Bytes()[0] != &payload[0] { t.Fatal("unexpected packet copy") }
    buf.ReleaseMulti(data)
    close(c.egress)
    if _,err := reader.ReadMultiBuffer(); !errors.Is(err,io.EOF) { t.Fatal("missing EOF") }
}

func TestAndroidUDPKeepsDestinationsSeparate(t *testing.T) {
    h := newUdpConnectionHandler(func(net.Conn,net.Destination){},func([]byte,net.Destination,net.Destination)error{return nil})
    src := net.UDPDestination(net.ParseAddress("172.19.0.1"),1000)
    first := net.UDPDestination(net.ParseAddress("1.1.1.1"),53)
    second := net.UDPDestination(net.ParseAddress("8.8.8.8"),53)
    h.HandlePacket(src,first,[]byte("one")); h.HandlePacket(src,second,[]byte("two"))
    if len(h.udpConns) != 2 { t.Fatal("different destinations shared one route") }
    for _, tc := range []struct{dst net.Destination; payload string}{{first,"one"},{second,"two"}} {
        c := h.udpConns[udpFlow{src,tc.dst}]
        var b [16]byte; n,err := c.Read(b[:])
        if err != nil || string(b[:n]) != tc.payload { t.Fatal("wrong flow payload") }
    }
    old := h.udpConns[udpFlow{src,first}]; old.Close()
    h.HandlePacket(src,first,[]byte("new")); old.Close()
    if len(h.udpConns) != 2 { t.Fatal("stale close removed replacement") }
}

func TestAndroidUDPCloseCanRacePackets(t *testing.T) {
    h := newUdpConnectionHandler(func(c net.Conn,_ net.Destination){c.Close()},func([]byte,net.Destination,net.Destination)error{return nil})
    src := net.UDPDestination(net.ParseAddress("172.19.0.1"),1000)
    dst := net.UDPDestination(net.ParseAddress("1.1.1.1"),53)
    var wg sync.WaitGroup
    for range 8 { wg.Go(func(){ for range 1000 { h.HandlePacket(src,dst,[]byte("packet")) } }) }
    wg.Wait()
}

func TestAndroidUDPWriterReleasesBuffersAndReturnsErrors(t *testing.T) {
    expected := errors.New("network closed")
    h := newUdpConnectionHandler(func(net.Conn,net.Destination){},func([]byte,net.Destination,net.Destination)error{return expected})
    dst := net.UDPDestination(net.ParseAddress("1.1.1.1"),53)
    c := &udpConn{handler:h,src:dst,dst:dst}
    if _,err := c.Write([]byte("data")); !errors.Is(err,expected) { t.Fatal("write error was swallowed") }
    b := buf.New(); b.WriteString("data")
    c.WriteMultiBuffer(buf.MultiBuffer{b})
    if b.Len() != 0 { t.Fatal("packet buffer not released") }
}
